1//! Per-thread compiled-plan cache behind [`crate::Tree::lookup`].
2//!
3//! A direct-mapped, 16-entry cache keyed by
4//! `(arena, target, source, generation)`. A topology mutation bumps the
5//! generation, so a stale plan can never be served: the key simply no longer
6//! matches and the entry is recompiled. The cache is `thread_local!`, so it
7//! needs `std` and lives in the facade, not the `no_std` core.
8//!
9//! **The arena component is load-bearing** (issue #196). One thread's cache is
10//! shared by every `Tree` it touches, and the other three components collide
11//! across trees as a matter of course: `FrameId`s are handed out in interning
12//! order, and a freshly built tree's generation is **its declared edge count**
13//! — one tick per link, measured, not zero — so two trees built from the same
14//! names in the same order agree on all three. A plan is edge indices plus the
15//! static transforms folded along the way, so serving one across trees returns
16//! the other tree's numbers where the topologies match and a number belonging
17//! to neither where they do not.
18//!
19//! That the generation is the edge count rather than a constant is why the
20//! defect is *narrower* than "any two trees": two trees with different edge
21//! counts miss on the generation and never reach the wrong plan at all. It is
22//! also why the narrowing is worth nothing as a defence — the sibling cases
23//! that matter (a tree rebuilt in a loop, two robots of the same model, a
24//! test suite's fixtures) are exactly the ones whose edge counts agree.
25//! `tests/plan_cache_identity.rs` is the three shapes of that.
26//!
27//! `docs/API.md` §1 R1 permits `lookup` to collapse the three tiers on the
28//! condition that it goes through this cache; R2's "never allocates, never
29//! locks" governs the *hot* tier — `Plan::at` — and this probe is neither an
30//! allocation nor a lock in any case.
31

38/// Number of direct-mapped slots. A power of two so indexing is a mask.
39const SLOTS: usize = 16;
40
41/// The odd multiplier [`index`] folds with.
42///
43/// At module scope rather than inside [`index`] because
44/// [`tests::the_low_bit_mask_wins_on_a_small_tree_and_ties_on_a_large_one`] builds the
45/// *rejected* alternative out of it: a private copy there would keep comparing
46/// the retuned shipped constant against a baseline built from the old one, and
47/// the comparison would stay green while measuring two different functions.
48///
49/// Its final digit is load-bearing — see [`index`].
50const MIX: u64 = 0x9E37_79B9_7F4A_7C15;
51
52/// What a cached plan was compiled from and for.
53#[derive(Clone, Copy, PartialEq, Eq)]
54struct Key {
55    /// Which arena the plan is meaningful against (`Tree::cache_scope`).
56    scope: u64,
57    target: u32,
58    source: u32,
59    generation: u64,
60}
61
62#[derive(Clone, Copy)]
63struct Entry {
64    key: Key,
65    plan: Plan,
66}
67
68thread_local! {
69    static CACHE: RefCell<[Option<Entry>; SLOTS]> = const { RefCell::new([None; SLOTS]) };
70}

72/// Map a key to its direct-mapped slot.
73///
74/// The fold seeds on the arena id and ends by masking the **low** bits, which
75/// is what the pre-#196 version did and is kept deliberately. Masking the low
76/// bits of a multiply looks like the weak choice — the low bits of a product
77/// depend only on the low bits of its operands, so the slot is a function of
78/// the low four bits of each component alone — and taking the high bits of a
79/// final multiply instead was written and measured first. **The measurement
80/// refuted it.** `0x9E37_79B9_7F4A_7C15` ends in `5`, and multiplication by 5
81/// modulo 16 is a bijection, so the mask is a *permutation* of the low bits of
82/// the key rather than a hash of them: keys that differ in those bits are
83/// guaranteed distinct slots, where a hash scatters them at random and collides
84/// at the birthday rate.
85///
86/// Steady-state residency in this 16-entry cache (fraction of a working set
87/// whose slot no other member shares), 2000 random working sets each:
88///
89/// | working set | this mask | high bits of a final multiply |
90/// |---|---|---|
91/// | 1 tree, 6 pairs, 8 frames | **0.719** | 0.513 |
92/// | 1 tree, 6 pairs, 40 frames | 0.717 | 0.726 |
93/// | 1 tree, 16 pairs, 40 frames | 0.384 | 0.378 |
94/// | 4 trees, 3 pairs, 8 frames | 0.493 | 0.505 |
95/// | 2 trees, 8 pairs, 40 frames | 0.384 | 0.381 |
96///
97/// [`tests::the_low_bit_mask_wins_on_a_small_tree_and_ties_on_a_large_one`]
98/// re-measures the first two rows in-crate, so the choice is not defended by a
99/// number in a comment alone; it draws its own working sets and reads 0.722
100/// against 0.510, then 0.717 against 0.726. It asserts only the *relative*
101/// claim in each — the win, and the tie — because the absolute figures are
102/// tuning, and a test that pins tuning fails on a retune that regressed
103/// nothing.
104///
105/// The mask wins where a tree is small enough for its ids to fit in the mask
106/// and ties elsewhere; one pair looked up across 16 consecutive arena ids lands
107/// in 16 distinct slots under the mask and 12 under the hash. The hash's only
108/// win is a population whose ids are all congruent modulo 16 — twelve frames at
109/// 3, 19, 35, … collapse onto one slot here — which a dense, interning-order id
110/// counter does not produce.
111///
112/// None of this is a correctness argument: the slot decides only whether a
113/// lookup hits, and the full [`Key`] comparison decides what it is allowed to
114/// return.
115fn index(key: Key) -> usize {
116    let mut h = key.scope;
117    h = h.wrapping_mul(MIX) ^ u64::from(key.target);
118    h = h.wrapping_mul(MIX) ^ u64::from(key.source);
119    h = h.wrapping_mul(MIX) ^ key.generation;
120    (h as usize) & (SLOTS - 1)
121}

123/// Evaluate `f` against the cached plan for
124/// `(tree's arena, target, source, generation)`, compiling and caching it first
125/// on a miss. The `bool` is `true` on a cache hit (used by tests).
126///
127/// The arena identity is read from `tree` rather than passed in, so no caller
128/// can supply one that does not belong to the tree it is about to compile from.
129///
130/// # Why a closure and not `-> Plan`
131///
132/// `Plan` is `Copy` and **2112 bytes** (`size_of`, measured; `align_of` 64,
133/// which is what pads the 2136-byte `Entry` out to a 2176-byte slot). Returning
134/// one hands the caller a copy, and the caller is `Tree::lookup`, which only
135/// wants to call `Plan::at` on it.
136///
137/// **Counted, not read off a disassembly.** A `memcpy` interposer
138/// (`LD_PRELOAD`, versioned `memcpy@GLIBC_2.14`) over a hot-cache harness, with
139/// the totals differenced between 1000 and 2000 lookups so process start-up
140/// cancels exactly. Three builds of one harness, because **two things had to
141/// change together and either one alone buys nothing**:
142///
143/// | probe binding | returns | copies per cache hit | ns/lookup | vs. shipped, same session |
144/// |---|---|---|---|---|
145/// | `&slots[idx]` | `Plan` | 3 x 2072 B | 606.9-616.2 | 575.3-580.1 |
146/// | `slots[idx]` | closure | 1 x 2176 B | 616.4-623.4 | 580.5-589.6 |
147/// | `&slots[idx]` | closure | **none** | — | — |
148///
149/// Row 3 is what ships, so it has no row of its own: it is the right-hand
150/// column, re-measured against each rejected arm in that arm's own session. The
151/// 2072-byte copies in row 1 are the returned `Plan` travelling slot -> stack
152/// temporary -> caller; the single 2176-byte copy in row 2 is the whole `Entry`
153/// lifted out of the slot *before* the key is compared, to decide 24 bytes'
154/// worth of question. Only the shipped form has neither. What survives on the
155/// hit path is two 184-byte copies of the `Result` return, which were there
156/// before and are not the plan.
157///
158/// Timings are medians of 25 rounds of 200 000 calls, the two builds under
159/// comparison run alternately — 6 runs each for row 1, 5 for row 2 — and
160/// non-overlapping in both pairings. The shipped arm reads 575-580 in one
161/// session and 580-590 in the other, which is why every comparison is paired
162/// and alternating and why no cell here may be read against a cell from the
163/// other row: the paired difference is the claim, the absolute number is not
164/// (`PHASE5.md` §9.3).
165///
166/// # The `&` is load-bearing, and this comment used to say it was not
167///
168/// An earlier revision recorded that `slots[idx]` and `&slots[idx]` compile to
169/// byte-identical machine code because LLVM sinks the copy past the key
170/// comparison, and concluded the reference was cosmetic. That was measured on
171/// the `-> Plan` shape, where it is beside the point: a function returning
172/// `Plan` by value copies the entry regardless, so removing one copy changes
173/// nothing. Once the return became a closure the copy had nowhere else to go,
174/// and row 2 is what it costs — a full slot memcpy per hit and the entire win
175/// gone. Do not "simplify" the `&` away.
176///
177/// # `f` cannot re-enter the cache
178///
179/// On a hit it runs while the `RefCell` is immutably borrowed, so a re-entrant
180/// lookup that *missed* would panic in `borrow_mut`. Today that is not a
181/// convention to be careful about but a property of the crate graph: the one
182/// caller's closure does `Tree::guard` then `Plan::at`, and `Guard` and `Plan`
183/// live in `tf_tree_core`, which `tf_tree` depends on and which therefore cannot
184/// name this module. A `pub(crate)` with one caller is the other half — the
185/// thing to re-check is a **second** caller whose closure reaches back into
186/// `tf_tree`, because that one would compile.
187///
188/// # Errors
189///
190/// Any [`LookupError`] from compilation on a miss.
191pub(crate) fn with_plan<R>(
192    tree: &Tree,
193    target: FrameId,
194    source: FrameId,
195    generation: u64,
196    f: impl FnOnce(&Plan) -> R,
197) -> Result<(R, bool), LookupError> {
198    let key = Key {
199        scope: tree.cache_scope(),
200        target: target.get(),
201        source: source.get(),
202        generation,
203    };
204    let idx = index(key);
205    CACHE.with(|c| {
206        {
207            let slots = c.borrow();
208            if let Some(entry) = &slots[idx] {
209                if entry.key == key {
210                    return Ok((f(&entry.plan), true));
211                }
212            }
213        }
214        let plan = tree.plan(target, source)?;
215        c.borrow_mut()[idx] = Some(Entry { key, plan });
216        Ok((f(&plan), false))
217    })
218}

226    /// The [`super::index`] mask keeps more of a small tree's working set
227    /// resident than the obvious alternative, taking the high bits of a final
228    /// multiply. This is the measurement behind that function's doc comment,
229    /// and the reason the pre-#196 low-bit mask was kept when the key grew.
230    ///
231    /// A key hits in steady state iff no other key in the working set maps to
232    /// its slot, so the metric is the fraction of a working set whose slot is
233    /// its own, averaged over 2000 synthetic sets from a fixed seed.
234    ///
235    /// **Two rows, and the second one is the honest one.** The mask wins where
236    /// a tree's ids fit inside the four bits it keeps and *ties* once they do
237    /// not, so a test carrying only the winning row would read as a claim that
238    /// the mask is better everywhere. Neither assertion is absolute: pinning
239    /// "the mask scores 0.72" would fail the build on a retune of [`super::MIX`]
240    /// or [`super::SLOTS`] that regressed nothing, and pinning "the alternative
241    /// scores under 0.55" would make the *rejected* arm part of the contract.
242    /// The claim is relative, per row, which is the claim the choice rests on.
243    ///
244    /// The alternative arm reads [`super::MIX`] rather than declaring its own
245    /// copy. With a copy, changing the shipped constant compared the new mask
246    /// against a baseline built from the old one — two different functions,
247    /// green either way.
248    #[test]
249    fn the_low_bit_mask_wins_on_a_small_tree_and_ties_on_a_large_one() {
250        // The alternative arm, not the shipped one: `super::index`'s fold with
251        // one more multiply, taking the top four bits instead of the bottom.
252        fn hashed(key: super::Key) -> usize {
253            let mut h = key.scope;
254            h = h.wrapping_mul(super::MIX) ^ u64::from(key.target);
255            h = h.wrapping_mul(super::MIX) ^ u64::from(key.source);
256            h = h.wrapping_mul(super::MIX) ^ key.generation;
257            (h.wrapping_mul(super::MIX) >> (u64::BITS - super::SLOTS.trailing_zeros())) as usize
258        }
259
260        // Mean residency of a `pairs`-pair working set over one tree of
261        // `frames` frames, as (mask, alternative).
262        let residency = |frames: u32, pairs: usize| {
263            // xorshift64, so the working sets are the same on every host and
264            // every run — a flaky instrument would be worse than no
265            // measurement.
266            let mut state = 0x1234_5678_9ABC_DEF1u64;
267            let mut next = move || {
268                state ^= state << 13;
269                state ^= state >> 7;
270                state ^= state << 17;
271                state
272            };
273            let resident = |slots: &[usize]| {
274                let mut counts = [0usize; super::SLOTS];
275                for &s in slots {
276                    counts[s] += 1;
277                }
278                slots.iter().filter(|&&s| counts[s] == 1).count() as f64 / slots.len() as f64
279            };
280            let (mut mask_total, mut hash_total) = (0.0, 0.0);
281            let trials = 2000;
282            for _ in 0..trials {
283                let (mut mask_slots, mut hash_slots) = (Vec::new(), Vec::new());
284                for _ in 0..pairs {
285                    let key = super::Key {
286                        scope: 1,
287                        target: (next() % u64::from(frames)) as u32,
288                        source: (next() % u64::from(frames)) as u32,
289                        generation: 0,
290                    };
291                    mask_slots.push(super::index(key));
292                    hash_slots.push(hashed(key));
293                }
294                mask_total += resident(&mask_slots);
295                hash_total += resident(&hash_slots);
296            }
297            (
298                mask_total / f64::from(trials),
299                hash_total / f64::from(trials),
300            )
301        };
302
303        // Eight frames: every id fits in the mask's four bits, so the mask is a
304        // permutation where the alternative is a hash colliding at the birthday
305        // rate. Measured 0.722 against 0.510.
306        let (mask, hash) = residency(8, 6);
307        assert!(
308            mask > hash + 0.15,
309            "on an 8-frame tree the mask {mask} should beat the alternative {hash} \
310             by the margin the choice was made on"
311        );
312        // Forty frames: the ids no longer fit, and the two are the same
313        // instrument. Measured 0.717 against 0.726 — the alternative is ahead,
314        // by less than the third digit of either.
315        let (mask, hash) = residency(40, 6);
316        assert!(
317            (mask - hash).abs() < 0.05,
318            "on a 40-frame tree the mask {mask} and the alternative {hash} tie; \
319             a gap either way means the index changed shape, not tuning"
320        );
321    }

323    /// Two trees on one thread do not evict each other's entries: each still
324    /// hits on a repeat, and they hold distinct arena ids.
325    ///
326    /// The fix for #196 is a fix only if the cache still caches. Widening the
327    /// key so far that nothing ever matches — or dropping the cache — would
328    /// answer every one of `tests/plan_cache_identity.rs`'s cases correctly and
329    /// be worth nothing, and would also breach `docs/API.md` §1 R1, which
330    /// permits `lookup` to collapse the three tiers *on the condition* that it
331    /// goes through this cache rather than re-resolving topology per call.
332    #[test]
333    fn two_trees_keep_separate_entries_and_still_hit() {
334        let build = || {
335            TreeBuilder::new()
336                .static_edge("a", "b", &Iso3::IDENTITY)
337                .build()
338                .unwrap()
339        };
340        let first = build();
341        let second = build();
342        assert_ne!(
343            first.cache_scope(),
344            second.cache_scope(),
345            "two heap trees are two arenas"
346        );
347
348        let key_of = |t: &crate::Tree| {
349            let a = t.frame("a").unwrap();
350            let b = t.frame("b").unwrap();
351            (a, b, t.guard().generation())
352        };
353        let (a1, b1, g1) = key_of(&first);
354        let (a2, b2, g2) = key_of(&second);
355        // The precondition for the whole bug: everything except the arena id
356        // agrees between the two trees.
357        assert_eq!((a1.get(), b1.get(), g1), (a2.get(), b2.get(), g2));
358
359        let probe = |t: &crate::Tree, target, source, g| {
360            super::with_plan(t, target, source, g, |_| ()).unwrap().1
361        };
362        assert!(!probe(&first, b1, a1, g1));
363        assert!(probe(&first, b1, a1, g1), "the first tree's repeat hits");
364        assert!(
365            !probe(&second, b2, a2, g2),
366            "the second tree must not be served the first tree's plan"
367        );
368        assert!(probe(&second, b2, a2, g2), "the second tree's repeat hits");
369    }

371    /// A repeated lookup hits the cache; a topology change (new generation)
372    /// produces a freshly-compiled plan stamped with the new generation.
373    #[test]
374    fn cache_hits_and_invalidates_on_generation() {
375        // `c` is declared with an edge (under `b`) so it can be re-parented at
376        // runtime to bump the topology generation.
377        let tree = TreeBuilder::new()
378            .static_edge("a", "b", &Iso3::IDENTITY)
379            .static_edge("b", "c", &Iso3::IDENTITY)
380            .build()
381            .unwrap();
382        let a = tree.frame("a").unwrap();
383        let b = tree.frame("b").unwrap();
384        let c = tree.frame("c").unwrap();
385
386        let gen1 = tree.guard().generation();
387        let (g1_stamped, hit1) =
388            super::with_plan(&tree, b, a, gen1, tf_tree_core::Plan::generation).unwrap();
389        assert!(!hit1, "first compile is a miss");
390        assert_eq!(g1_stamped, gen1);
391
392        // An immediate repeat with the same key hits the per-thread cache.
393        let (_, hit2) = super::with_plan(&tree, b, a, gen1, |_| ()).unwrap();
394        assert!(hit2, "repeat lookup hits the cache");
395
396        // A runtime re-parent bumps the generation; the recompiled plan carries it.
397        tree.reparent(c, a).unwrap();
398        let gen2 = tree.guard().generation();
399        assert_ne!(gen1, gen2, "re-parent must change the generation");
400
401        let (g3_stamped, _hit3) =
402            super::with_plan(&tree, b, a, gen2, tf_tree_core::Plan::generation).unwrap();
403        assert_eq!(
404            g3_stamped, gen2,
405            "post-change plan is stamped with the new generation"
406        );
407    }

409    /// The cache still caches, measured on **live trees** rather than on a model
410    /// of the index function — and measured against what the index *predicts*
411    /// for the ids those trees actually got, not against a fixed number.
412    ///
413    /// [`the_low_bit_mask_wins_on_a_small_tree_and_ties_on_a_large_one`] argues
414    /// the *choice* of index from synthetic keys; this measures the thing that
415    /// choice is for. N trees, round-robin, three rounds, counting only the
416    /// rounds after the first — the steady state a real reader is in, which one
417    /// warm round is enough to reach.
418    ///
419    /// # Why it does not assert a hit rate
420    ///
421    /// The obvious form — "N <= 16 live trees must hit 100% of the time" — is
422    /// true only while `next_local_scope`'s counter hands *these* trees
423    /// consecutive ids, and that counter is process-global. `just miri` runs
424    /// `-p tf_tree --lib` on libtest's default multi-threaded harness, so a
425    /// sibling test building a tree between two of these builds leaves a gap;
426    /// two of sixteen ids then agree modulo 16, evict each other, and the rate
427    /// is 0.875 through no fault of the cache. `nextest` gives every test its
428    /// own process and would never show it, so the miri job would be the only
429    /// one to flap — the worst place for a flake to live.
430    ///
431    /// So the expectation is **derived from the ids that were minted**: the
432    /// slots are recomputed through [`super::index`] and a tree is expected to
433    /// hit iff no other tree in the set shares its slot. Whatever else is
434    /// running, the assertion is exact rather than tolerant, and it is a
435    /// stronger statement than the rate ever was: the cache hits *exactly* when
436    /// its index says it should.
437    ///
438    /// # What that catches
439    ///
440    /// * The #196 defect itself, at every N including 2: a key that stopped
441    ///   separating trees makes every tree share one entry, so the measured
442    ///   hits go to `total` while the prediction goes to zero.
443    /// * An eviction or install bug, from the other side: no hits at all
444    ///   against a prediction of N.
445    /// * An index that stopped being a bijection on the low bits of `scope` —
446    ///   the property that keeps two heap trees from thrashing, and the
447    ///   performance regression this fix could plausibly have introduced. That
448    ///   is the second assertion: slot-uniqueness must agree with
449    ///   `scope`-modulo-[`super::SLOTS`] uniqueness, tree for tree.
450    ///
451    /// N = 17 is kept because the pigeonhole is the one thing no key can talk
452    /// its way out of: seventeen trees cannot all be resident in sixteen slots,
453    /// so a run where they are is a run where the arena component is not in the
454    /// key.
455    #[test]
456    fn the_cache_hits_exactly_where_its_index_predicts() {
457        let build = || {
458            TreeBuilder::new()
459                .static_edge("a", "b", &Iso3::IDENTITY)
460                .static_edge("b", "c", &Iso3::IDENTITY)
461                .build()
462                .unwrap()
463        };
464        // How many members of `slots` no other member collides with.
465        let resident = |slots: &[usize]| {
466            slots
467                .iter()
468                .filter(|&&s| slots.iter().filter(|&&o| o == s).count() == 1)
469                .count()
470        };
471
472        const ROUNDS: usize = 3;
473        for n in [2usize, 16, 17] {
474            let trees: Vec<crate::Tree> = (0..n).map(|_| build()).collect();
475
476            // Every tree interned the same names in the same order and declared
477            // the same edges, so the key's other three components agree across
478            // the set and the slot is a function of `scope` alone. That is the
479            // #196 precondition, and it is asserted rather than assumed.
480            let a = trees[0].frame("a").unwrap();
481            let c = trees[0].frame("c").unwrap();
482            let g = trees[0].guard().generation();
483            for t in &trees {
484                assert_eq!(
485                    (
486                        t.frame("a").unwrap().get(),
487                        t.frame("c").unwrap().get(),
488                        t.guard().generation()
489                    ),
490                    (a.get(), c.get(), g),
491                    "the trees must agree on everything but their arena id"
492                );
493            }
494
495            let slots: Vec<usize> = trees
496                .iter()
497                .map(|t| {
498                    super::index(super::Key {
499                        scope: t.cache_scope(),
500                        target: a.get(),
501                        source: c.get(),
502                        generation: g,
503                    })
504                })
505                .collect();
506            let residues: Vec<usize> = trees
507                .iter()
508                .map(|t| (t.cache_scope() as usize) & (super::SLOTS - 1))
509                .collect();
510
511            let (mut hits, mut total) = (0usize, 0usize);
512            for round in 0..ROUNDS {
513                for tree in &trees {
514                    let (_, hit) = super::with_plan(tree, a, c, g, |_| ()).unwrap();
515                    if round > 0 {
516                        total += 1;
517                        hits += usize::from(hit);
518                    }
519                }
520            }
521
522            assert_eq!(
523                hits,
524                resident(&slots) * (ROUNDS - 1),
525                "{n} trees: the cache hit {hits} times in {total} steady-state \
526                 lookups, but its own index puts {} of them in a slot no other \
527                 tree shares",
528                resident(&slots)
529            );
530            assert_eq!(
531                resident(&slots),
532                resident(&residues),
533                "{n} trees: `index` must separate arena ids exactly as their low \
534                 {} bits do — it is a permutation of them, and that is what \
535                 keeps consecutive ids from thrashing",
536                super::SLOTS.trailing_zeros()
537            );
538            if n > super::SLOTS {
539                assert!(
540                    hits < total,
541                    "{n} trees cannot all be resident in {} slots — {hits} of \
542                     {total} means the arena component stopped separating them",
543                    super::SLOTS
544                );
545            }
546        }
547    }

549    /// Two handles onto **one shared segment** share one arena identity, and
550    /// therefore one set of cached plans.
551    ///
552    /// This is the other half of the fix and the half a test could easily get
553    /// backwards. Keying on the *handle* would answer every case in
554    /// `tests/plan_cache_identity.rs` correctly and still be wrong here: two
555    /// `Tree`s mapping one segment see one topology and one set of static
556    /// transforms, so a plan compiled through either is correct through the
557    /// other, and giving them separate identities would cost every second
558    /// handle a recompile for no safety. The peer's *first* lookup hitting the
559    /// owner's entry is the assertion that pins that.
560    ///
561    /// Measured here: owner and peer both report the same tagged scope, and the
562    /// peer's first [`super::with_plan`] returns `hit = true`.
563    ///
564    /// Reachable only through `just shm-check`'s
565    /// `cargo nextest run -p tf_tree --features shm --lib` line, which was added
566    /// with this test — `just test` builds default features, so without that
567    /// line this would be compiled by clippy and executed by nothing.
568    ///
569    /// **No `TF_TREE_RUNTIME_DIR` scratch directory here, deliberately**, unlike
570    /// `tests/rendezvous.rs`, `tests/owned_writer.rs` and
571    /// `tf_tree_cli/tests/attach.rs`. Those go through `tf_tree::Open`, whose
572    /// rendezvous *is* a lock file and a socket under that directory, so two
573    /// concurrent runs collide and a killed run leaves a lock file behind.
574    /// `build_shared` reaches none of it: it is `memfd_create` plus `mmap`
575    /// (`tf_tree_arena::MappedArena::create`), the fd is the only capability,
576    /// and the name is a debug label that shows up in `/proc/<pid>/fd` — memfd
577    /// names are not unique and not a namespace. Measured: this test passes with
578    /// `TF_TREE_RUNTIME_DIR` pointed at a path that does not exist, and eight
579    /// copies of it run concurrently all pass.
580    #[cfg(all(feature = "shm", target_os = "linux"))]
581    #[test]
582    fn two_handles_on_one_shared_arena_share_their_plans() {
583        let owner = TreeBuilder::new()
584            .static_edge("a", "b", &Iso3::IDENTITY)
585            .build_shared("tf_tree-cache-identity-test")
586            .unwrap();
587        let fd = owner
588            .shared_fd()
589            .expect("a build_shared tree has a segment fd")
590            .try_clone_to_owned()
591            .unwrap();
592        let peer = crate::Tree::attach_shared(fd, crate::AttachMode::ReadOnly).unwrap();
593
594        assert_eq!(
595            owner.cache_scope(),
596            peer.cache_scope(),
597            "one segment is one arena; both handles must key the same"
598        );
599        assert_eq!(
600            owner.cache_scope() >> 63,
601            1,
602            "a shared scope carries the tag bit that keeps it out of the counter's space"
603        );
604
605        let a = owner.frame("a").unwrap();
606        let b = owner.frame("b").unwrap();
607        let g = owner.guard().generation();
608        assert!(
609            !super::with_plan(&owner, b, a, g, |_| ()).unwrap().1,
610            "cold cache"
611        );
612        assert!(
613            super::with_plan(&peer, b, a, g, |_| ()).unwrap().1,
614            "the peer's FIRST lookup must reuse the owner's plan, not recompile"
615        );
616    }
